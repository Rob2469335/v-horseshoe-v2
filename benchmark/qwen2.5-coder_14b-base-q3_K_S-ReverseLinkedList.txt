class ListNode:
    def __init__(self, x):
        self.val = x
        self.next = None

def reverseLinkedList(head):
    prevNode = nextNode = None
        
    while head != None:
        # save the next node
        nextNode = head.next
        
        # change current's pointer direction to point back at previous 
        head.next = prevNode
        
        # shift pointers forwards by one position in list
        prevNode = head
        head = nextNode
    
    return prevNode
