class ListNode:
    def __init__(self, x):
        self.val = x
        self.next = None

def reverseList(head: ListNode) -> ListNode:
    prev = None
    current = head
    while current is not None:
        next_node = current.next  # Store next node
        current.next = prev       # Reverse the link
        prev = current            # Move pointers one position ahead.
        current = next_node
    return prev
