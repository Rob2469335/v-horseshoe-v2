class Node:
    def __init__(self, key, value):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}
        self.head = Node(0, 0)
        self.tail = Node(0, 0)
        self.head.next = self.tail
        self.tail.prev = self.head

    def _add_node(self, node):
        previous = self.tail.prev
        previous.next = node
        node.prev = previous
        node.next = self.tail
        self.tail.prev = node

    def _remove_node(self, node):
        previous = node.prev
        next = node.next
        previous.next = next
        next.prev = previous

    def get(self, key):
        if key in self.cache:
            node = self.cache[key]
            self._remove_node(node)
            self._add_node(node)
            return node.value
        else:
            return -1

    def set(self, key, value):
        if key in self.cache:
            self._remove_node(self.cache[key])
        elif len(self.cache) >= self.capacity:
            self._remove_node(self.tail.prev)
        self._add_node(Node(key, value))
        self.cache[key] = Node(key, value)

    def evict(self):
        node = self.head.next
        while node != self.tail:
            self._remove_node(node)
            del node
            node = self.head.next

def test_lru_cache():
    cache = LRUCache(3)
    assert cache.get('a') == -1
    cache.set('a', 10)
    assert cache.get('a') == 10
    cache.set('b', 20)
    cache.set('c', 30)
    assert cache.get('b') == 20
    cache.set('d', 40)
    assert cache.get('c') == -1
    assert cache.evict() is None
