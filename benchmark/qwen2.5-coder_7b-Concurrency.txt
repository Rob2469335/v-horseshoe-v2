```python
from threading import Lock

class ThreadSafeQueue:
    def __init__(self):
        self.queue = []
        self.lock = Lock()

    def enqueue(self, item):
        with self.lock:
            self.queue.append(item)

    def dequeue(self):
        with self.lock:
            if not self.queue:
                return None
            return self.queue.pop(0)
```
